\newcommand{\bookicon}[3]{\begin{scope}[shift={(#1,#2)},scale=#3]

\newcommand{\clockicon}[3]{\begin{scope}[shift={(#1,#2)}]
  \draw[art,fill=canvas] (0,0) circle (#3);

\end{scope}}
